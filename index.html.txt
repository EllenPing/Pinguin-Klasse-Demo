let aktuellerStundenTag = null;

function waehleStundenTag(tag, element){

  const tagGewechselt =
    aktuellerStundenTag !== null &&
    aktuellerStundenTag !== tag;

    document.querySelectorAll(".stundenErgebnis")

  }

  pruefeWoche();
}

/* WOCHENAUSWERTUNG */

function pruefeWoche(){

  const felder =
    document.querySelectorAll(".tag .ergebnis");

  let ausgefuellt = 0;
  let pinguine = 0;


  felder.forEach(function(feld){

    const symbol =
      feld.dataset.symbol || "";

    if(symbol !== ""){
      ausgefuellt++;
    }

    if(symbol === "🐧"){
      pinguine++;
    }

  });


  /* ERST NACH ALLEN 5 TAGEN */

  if(ausgefuellt < 5){
    return;
  }


  if(wocheSchonGewertet){
    return;
  }


  wocheSchonGewertet = true;


  /* MINDESTENS 3 PINGUINE */

  if(pinguine >= 3){

    erfolgreicheWochen++;


    document.getElementById("wochenZahl").textContent =
      erfolgreicheWochen;


    if(erfolgreicheWochen >= 1){

      document.getElementById("eisscholle").style.display =
        "block";
    }


    if(erfolgreicheWochen >= 2){

      document.getElementById("schollenPinguin").style.display =
        "block";
    }


    if(erfolgreicheWochen >= 3){

      document.getElementById("seerobbe").style.display =
        "block";
    }


    document.getElementById("wochenTitel").textContent =
      "Wir haben unser Wochenziel erreicht! ❄️";

    document.getElementById("wochenText").textContent =
      "";

  } else {

    document.getElementById("wochenTitel").textContent =
      "Wir bleiben dran! ❄️";
